\section{符号说明}
为便于表述，统一符号与单位如下（功率单位为 kW，能量单位为 kWh，价格单位为 元/kWh）：

\begin{description}
  \item[$T$] 天内时间步数，$T=144$（10 分钟步长）；$\Delta=\dfrac{1}{6}$ 小时。
  \item[$p_t$] 时段 $t$ 的电价（元/kWh）。
  \item[$L_t$] 时段 $t$ 的负荷功率（kW）；对应能量 $L_t^E=L_t\Delta$（kWh）。
  \item[$\widehat{PV}_t$] 时段 $t$ 的光伏预测功率（kW）；对应能量 $PV_t^E=\widehat{PV}_t\Delta$（kWh）。
  \item[$PV_t^{\mathrm{act}}$] 时段 $t$ 的光伏实际功率（kW）；对应能量 $PV_t^{\mathrm{act},E}=PV_t^{\mathrm{act}}\Delta$（kWh）。
  \item[$E_0$] 初始储能电量（kWh），题设 $E_0=6000$。
  \item[$E_{\min},E_{\max}$] 储能电量上下界（kWh），题设 $E_{\min}=1200,\ E_{\max}=10800$。
  \item[$P_{\mathrm{ch}}^{\max},P_{\mathrm{dis}}^{\max}$] 充/放电功率上界（kW），题设均为 $5000$。
  \item[$\eta_{\mathrm{c}},\eta_{\mathrm{d}}$] 充/放电效率，满足 $\eta_{\mathrm{c}}\eta_{\mathrm{d}}=\eta_{\mathrm{rt}}=0.9$（可取 $\eta_{\mathrm{c}}=\eta_{\mathrm{d}}=\sqrt{0.9}$）。
  \item[$G^p_t$] 计划购电量（kWh），在 0:00 规划得到并于对应时段从外网取电。
  \item[$G^d_t$] ``充电宝'' 服务/折扣通道购电量（kWh），单位电价为 $0.9\,p_t$（即打九折），可选并受最大可用量 $G^{d,\max}_t$ 限制。
  \item[$G^e_t$] 紧急/额外购电量（kWh），单位电价为 $5\,p_t$（高价惩罚，紧急补购）。
\end{description}

\subsection{决策变量（汇总）}
\begin{itemize}
  \item $G^p_t\ge0$：时段 $t$ 的计划购电量（kWh）。
  \item $G^d_t\ge0$：时段 $t$ 的充电宝购电量（kWh），单价为 $0.9p_t$，上限 $G^{d,\max}_t$（可设为常数或时间序列）。
  \item $G^e_t\ge0$：时段 $t$ 的紧急购电量（kWh），单价为 $5p_t$。
  \item $P_{\mathrm{ch},t}\ge0,\ P_{\mathrm{dis},t}\ge0$：时段 $t$ 的充/放电功率（kW），对应能量为 $P\Delta$（kWh）。
  \item $E_t$：时段 $t$ 末的储能电量（kWh）。
\end{itemize}

\section{问题1：确定性日内计划（模型建立与求解）}
问题1 与此前一致：在 0:00 已知预测数据 $\{p_t, L_t, \widehat{PV}_t\}$，制定当天的计划购电 $\{G^p_t\}$ 与储能充放电策略，最小化计划购电费用（不含折扣/紧急购电的事后成本）。本小节保持与竞赛题目一致的线性规划表述，供方法 B 事后仿真使用。下面给出模型与矩阵化求解提示。

\paragraph{目标函数（问题1，规划成本）}
\begin{equation}
\min_{G^p,P_{\mathrm{ch}},P_{\mathrm{dis}},E}\; J_1 \;=\; \sum_{t=1}^{T} p_t\,G^p_t.
\end{equation}

\paragraph{约束}
\begin{enumerate}
  \item 能量平衡（使用预测光伏）：
  \begin{equation}
  G^p_t + P_{\mathrm{dis},t}\Delta + PV_t^E = L_t^E + P_{\mathrm{ch},t}\Delta,\qquad t=1,\dots,T.
  \end{equation}
  \item 储能动力学：
  \begin{equation}
  E_t = E_{t-1} + \eta_{\mathrm{c}}(P_{\mathrm{ch},t-1}\Delta) - \frac{1}{\eta_{\mathrm{d}}}(P_{\mathrm{dis},t-1}\Delta),\qquad t=1,\dots,T.
  \end{equation}
  \item 边界条件：$E_{\min}\le E_t\le E_{\max}$；$0\le P_{\mathrm{ch},t}\le P_{\mathrm{ch}}^{\max}$；$0\le P_{\mathrm{dis},t}\le P_{\mathrm{dis}}^{\max}$。
  \item 非负性与循环终止：$G^p_t\ge0$，且 $E_T=E_0$（题设要求）。
\end{enumerate}

\paragraph{求解说明}
将问题1 写为标准的线性规划 $\min f^\top x$ s.t. $A_{\mathrm{eq}}x=b_{\mathrm{eq}},\ lb\le x\le ub$，变量向量 $x$ 可按 $[G^p_{1:T},\; P_{\mathrm{ch},1:T},\; P_{\mathrm{dis},1:T},\; E_{1:T}]^\top$ 排列。推荐在 MATLAB 中使用 \texttt{linprog} 求解（或商用求解器）。问题1 的解 $G^p_t$ 将作为问题2 的输入（方法 B）。

\section{问题2：包含四类供电来源的模型（完整规划与事后仿真）}
本节给出考虑四类供电来源的建模与求解方法：
\begin{enumerate}
  \item 计划购电（$G^p_t$），在 0:00 规划得到并按计划购入；单价 $p_t$。
  \item 光伏发电（预测或实际），$PV_t$，优先用于负荷/充电。
  \item 充电宝/折扣通道购电（$G^d_t$），单价为 $0.9\,p_t$（打 9 折），可能有时变上限 $G^{d,\max}_t$。
  \item 紧急/额外购电（$G^e_t$），单价为 $5\,p_t$，仅在不可用其他资源时发生。
\end{enumerate}

此处同时给出「规划内同时考虑充电宝和紧急购电」（方法 A）与「先规划后仿真、事后统计紧急与折扣购电量」（方法 B）的模型与求解建议。

\subsection{方法 A：规划内同时考虑所有来源（扩展 LP）}
在规划阶段把 $G^p_t,G^d_t,G^e_t$ 全部作为决策变量并计入目标函数，目标为规划期内总期望购电成本（采用预测 PV）：
\begin{equation}
\min\; J_{2A} \;=\; \sum_{t=1}^T \big( p_t G^p_t + 0.9p_t G^d_t + 5p_t G^e_t \big).
\label{eq:obj2A}
\end{equation}

约束（每时段）：
\begin{equation}
G^p_t + G^d_t + G^e_t + P_{\mathrm{dis},t}\Delta + PV_t^E = L_t^E + P_{\mathrm{ch},t}\Delta,\qquad t=1,\dots,T.
\label{eq:bal2A}
\end{equation}

此外：
\begin{align}
&0 \le G^d_t \le G^{d,\max}_t,\qquad G^p_t\ge0,\qquad G^e_t\ge0,\nonumber\\
&\text{储能动力学、容量与功率界同问题1。}
\end{align}

\paragraph{说明}
方法 A 的优点是规划阶段已权衡折扣通道与紧急购电成本，可能在总成本上更优；缺点是若对实际 PV 有较大不确定性，规划可能仍无法避免大量事后紧急购电。该模型仍为线性规划，可用 \texttt{linprog} 求解（变量向量包含 $G^p,G^d,G^e,P_{\mathrm{ch}},P_{\mathrm{dis}},E$）。

\subsection{方法 B：先规划（问题1）后仿真，统计事后购电}
方法 B 更贴合 "制定计划 + 运行时补购" 的实际流程，步骤如下：
\begin{enumerate}
  \item 在 0:00 用问题1 求得计划购电 $G^p_t$（并可得到一套储能调度策略 $P_{\mathrm{ch},t},P_{\mathrm{dis},t}$），不在规划阶段使用 $G^d$ 或 $G^e$。
  \item 在运行时（基于真实观测 $PV_t^{\mathrm{act}},L_t^{\mathrm{act}}$）按时序推进，对每个时段按优先级分配供电：
  \begin{enumerate}
    \item 优先使用光伏发电 $PV_t^{\mathrm{act}}$ 满足负荷或用于储能。
    \item 若仍有供电缺口，优先使用计划购电 $G^p_t$（已订购）。
    \item 若仍有缺口，则使用充电宝折扣通道 $G^d_t$（受上限 $G^{d,\max}_t$）；其费用按 $0.9p_t$ 计。
    \item 若仍有缺口，则以紧急价格购电 $G^e_t$（单价 $5p_t$）。
  \end{enumerate}
  \item 在仿真中，每时段计算实际发生的折扣购电 $G^{d,\mathrm{act}}_t$ 与紧急购电 $G^{e,\mathrm{act}}_t$，并记录其费用。事后总费用为：
  \begin{equation}
  \mathrm{Cost}^{\mathrm{act}} = \sum_{t=1}^T p_t G^p_t + \sum_{t=1}^T 0.9p_t G^{d,\mathrm{act}}_t + \sum_{t=1}^T 5p_t G^{e,\mathrm{act}}_t.
  \end{equation}
\end{enumerate}

\paragraph{逐时仿真计算流程（伪代码）}
\begin{enumerate}
  \item 读取 $G^p_t$（规划结果）、$P_{\mathrm{ch},t},P_{\mathrm{dis},t}$（若固定）、以及真实时序 $PV_t^{\mathrm{act}},L_t^{\mathrm{act}}$。
  \item 对每个时段 t：
  \begin{itemize}
    \item 计算需求能量 $D_t = L_t^{\mathrm{act}}\Delta + P_{\mathrm{ch},t}\Delta$（若按某策略充电）。
    \item 可用能量先为 $A_t = PV_t^{\mathrm{act}}\Delta + G^p_t + P_{\mathrm{dis},t}\Delta$（计划购电视为可用）。
    \item 若 $A_t\ge D_t$，则 $G^{d,\mathrm{act}}_t=0,\ G^{e,\mathrm{act}}_t=0$，并按规则更新 $E_t$。
    \item 否则令 $\mathrm{deficit}=D_t-A_t$。先从折扣通道取 $G^{d,\mathrm{act}}_t=\min(\mathrm{deficit},G^{d,\max}_t)$，若仍有不足，用紧急购电 $G^{e,\mathrm{act}}_t=\mathrm{deficit}-G^{d,\mathrm{act}}_t$。
  \end{itemize}
\end{enumerate}

\subsection{实现建议与输出}
\begin{itemize}
  \item MATLAB 实现：将方法 A 写为 LP 并调用 \texttt{linprog}；方法 B 在问题1 解的基础上编写逐时仿真脚本计算 $G^{d,\mathrm{act}},G^{e,\mathrm{act}}$ 与最终成本。
  \item 输出文件：生成 result1.xlsx（计划购电与储能调度）、result2.xlsx（问题2 的折扣购电与紧急购电记录与费用明细），并绘制时序图（负荷、PV、各类购电与 SoC）。
  \item 建议对关键参数（$G^{d,\max}$、折扣率、紧急价格倍数、$E_{\max}$）做敏感性分析以展示稳健性。
\end{itemize}
